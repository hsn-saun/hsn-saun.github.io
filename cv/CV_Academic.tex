\section{Research interests}
{\fontsize{9.8}{12.6}\selectfont
Human-aware and social robot navigation; multi-agent pedestrian trajectory
prediction; reinforcement learning for planning and control; human--robot
interaction and pedestrian comfort. I am interested in representations that
encode human state, social interaction structure and group behaviour, and in
validating them on physical robots rather than in simulation alone.}

\section{Research experience}

\entry{Feb 2024 -- present}{Research Fellow}{Networked Robotic Systems Laboratory, National Cheng Kung University}
\begin{achievements}
  \item Built an end-to-end human-aware navigation pipeline covering sensing,
        human-state representation, prediction, planning, learning and control,
        validated on an Agilex Scout Mini with ZED-X stereo vision, Velodyne
        VLP-16 LiDAR and Jetson AGX Orin compute.
  \item Developed Social Graph Mamba, an attention-free trajectory forecasting
        model using selective state space models over dynamic interaction graphs,
        reaching linear rather than quadratic complexity in the number of agents.
  \item Ran pedestrian studies with human participants, including comfort surveys
        and their statistical analysis, feeding empirical comfort models back into
        the navigation stack.
\end{achievements}

\entry{2022 -- Jan 2024}{Research Assistant and Team Leader}{Hanoi Intelligent Robotics Laboratory, Hanoi University of Industry}
\begin{achievements}
  \item Led a student research team on reinforcement learning for autonomous
        mobile robots, producing the group's first published navigation work.
  \item Implemented and benchmarked Q-Learning and SARSA navigation policies in
        Gazebo, with experimental validation on physical platforms.
\end{achievements}
